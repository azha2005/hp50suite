# Casos de DINVC: 'F' A B -> { FINV IMA IMB CA CB MSG }
#     nombre | expresion UserRPL | resultado esperado (como \->STR)
# ESPERADO calculados a mano ANTES de correr. La inversa se verifica
# evaluandola en un punto (no depende de como escribe el CAS).

USAR: DINVC

# 2022 T1 ej 4: f:[-2,+oo)->B, (x+2)^2-4. B=[-4,+oo), f^-1=-2+raiz(x+4)
inv-t1-f      | '(X+2)^2-4' -2 "INF" DINVC 1 GET 'X' 5 = SUBST \->NUM | 1.
inv-t1-a      | '(X+2)^2-4' -2 "INF" DINVC 2 GET        | -4
inv-t1-b      | '(X+2)^2-4' -2 "INF" DINVC 3 GET        | '+\oo'
inv-t1-ca     | '(X+2)^2-4' -2 "INF" DINVC 4 GET        | 1
inv-t1-cb     | '(X+2)^2-4' -2 "INF" DINVC 5 GET        | 0
inv-t1-msg    | '(X+2)^2-4' -2 "INF" DINVC 6 GET        | OK
# 2022 T2 y T3
inv-t2-f      | '(X-2)^2-4' 2 "INF" DINVC 1 GET 'X' 5 = SUBST \->NUM | 5.
inv-t3-f      | '(X+1)^2-1' -1 "INF" DINVC 1 GET 'X' 3 = SUBST \->NUM | 1.
# Primer parcial(1): (x-1)^2+2 -> [1,+oo) -> [2,+oo)
inv-pp1-f     | '(X-1)^2+2' 1 "INF" DINVC 1 GET 'X' 6 = SUBST \->NUM | 3.
inv-pp1-a     | '(X-1)^2+2' 1 "INF" DINVC 2 GET         | 2
# 2023 T4 ej 4: raiz(x-4) -> [4,+oo) -> [0,+oo), f^-1 = x^2+4
inv-raiz-f    | '\v/(X-4)' 4 "INF" DINVC 1 GET 'X' 3 = SUBST \->NUM | 13.
inv-raiz-a    | '\v/(X-4)' 4 "INF" DINVC 2 GET          | 0
# Recu 2025 ej 3: ln(x+1) -> (-1,+oo) -> R, f^-1 = e^x - 1
inv-ln-f      | 'LN(X+1)' -1 "INF" DINVC 1 GET 'X' 1 = SUBST \->NUM 4 RND | 1.7183
inv-ln-a      | 'LN(X+1)' -1 "INF" DINVC 2 GET          | '-\oo'
inv-ln-ca     | 'LN(X+1)' -1 "INF" DINVC 4 GET          | 0
# 2017: 2^x+1 : R -> (1,+oo), f^-1 = ln(x-1)/ln 2
inv-exp-f     | '2^X+1' "-INF" "INF" DINVC 1 GET 'X' 3 = SUBST \->NUM | 1.
inv-exp-a     | '2^X+1' "-INF" "INF" DINVC 2 GET        | 1
inv-exp-ca    | '2^X+1' "-INF" "INF" DINVC 4 GET        | 0
# 2024 T2 ej 4: x^2-1 : [-3,0] -> [-1,8], decreciente, f^-1 = -raiz(x+1)
inv-dec-f     | 'X^2-1' -3 0 DINVC 1 GET 'X' 3 = SUBST \->NUM | -2.
inv-dec-a     | 'X^2-1' -3 0 DINVC 2 GET                | -1
inv-dec-b     | 'X^2-1' -3 0 DINVC 3 GET                | 8
inv-dec-cb    | 'X^2-1' -3 0 DINVC 5 GET                | 1
# 2014: |2x-4| en [2,+oo), f^-1 = (x+4)/2
inv-abs-f     | 'ABS(2*X-4)' 2 "INF" DINVC 1 GET 'X' 6 = SUBST \->NUM | 5.
# No inyectiva en todo R
inv-noiny     | 'X^2' "-INF" "INF" DINVC 6 GET 1 12 SUB  | NO INYECTIVA
# X^3: f'(0)=0 pero no da la vuelta, es inyectiva
inv-cubo      | 'X^3' "-INF" "INF" DINVC 6 GET          | OK
inv-depth     | DEPTH \-> D0 \<< 'X^2' "-INF" "INF" DINVC DROP DEPTH D0 - \>> | 0
