# Laboratorio de Datos: los nucleos de calculo.
#     nombre | expresion UserRPL | esperado
# Predicciones calculadas en la PC antes de correr (ver el mensaje del commit).
# Las de DPCTIL salen del apunte de catedra (AGENTS.md seccion 6).

# DOLSC ya no calcula las sumatorias: llama a DCOVCORRC, asi que hay que
# cargar las dos. Si faltara, DOLSC dejaria el nombre 'DCOVCORRC' sin evaluar
# en la pila en vez de dar error.
USAR: DSTATSC DPCTILC DMODAC DOLSC DCOVCORRC DFRECTABLAC DPIBPPAC DTCAMBIOC DCOMPZC
USAR: DMKTC DCOMPFIRMC DBOXPLOTC

# ---------- DSTATSC: Datos -> { N XBAR VARZ DESV CV } ----------
# {2,4,6,8,10}: media 6, varianza muestral 10, desvio raiz(10).
stats-lista     | { 2 4 6 8 10 } DSTATSC                     | { 5. 6. 10. 3.16227766017 52.7046276695 }
# Mismo dato como VECTOR: tiene que dar identico. Ojo: en modo EXACTO un
# vector de enteros SIN punto no es TYPE 3 sino TYPE 29 (array simbolico),
# medido en casos-arr.txt. Es lo que tipea un alumno, y antes daba ERROR.
stats-vector    | [ 2 4 6 8 10 ] DSTATSC 2 GET               | 6.
stats-vector-r  | [ 2. 4. 6. 8. 10. ] DSTATSC 2 GET          | 6.
pctil-vector    | [ 2 4 6 8 10 ] 70 DPCTILC 3 GET            | 8.4
moda-vector     | [ 1 2 2 3 ] DMODAC 2 GET                   | 2.
frec-vector     | [ 10 12 14 16 ] 2 0 0 DFRECTABLAC 1 GET     | 2.
boxplot-vector  | [ 2 4 6 8 10 ] DBOXPLOTC 4 GET             | 6.
covcorr-vector  | [ 1 2 3 ] [ 2 4 6 ] DCOVCORRC 1 GET        | 3.

# ---------- DPCTILC: Datos P -> { N LP VAL } ----------
# Caso del apunte: p50 -> 6; p70 -> Lp=4.2, Valor=8.4.
pctil-mediana   | { 2 4 6 8 10 } 50 DPCTILC                  | { 5. 3. 6. }
pctil-p70       | { 2 4 6 8 10 } 70 DPCTILC                  | { 5. 4.2 8.4 }
# Desordenado: DPCTILC ordena solo, tiene que dar lo mismo.
pctil-desorden  | { 10 2 8 4 6 } 70 DPCTILC 3 GET            | 8.4

# ---------- DMODAC: Datos -> { MODES MAXFREQ } ----------
moda-unica      | { 1 2 2 3 3 3 } DMODAC                     | { { 3 } 3 }
moda-bimodal    | { 1 1 2 2 3 } DMODAC                       | { { 1 2 } 2 }
# Todos distintos: no hay moda, y eso se ve en MAXFREQ = 1.
moda-sin        | { 1 2 3 } DMODAC 2 GET                     | 1

# ---------- DOLSC: X Y -> { N B0 B1 R R2 } ----------
# Perfectamente lineal: Y = 1 + 2X. B0 y B1 tienen que dar exactos.
ols-perfecta-b  | { 1 2 3 4 } { 3 5 7 9 } DOLSC 2 GET        | 1.
ols-perfecta-b1 | { 1 2 3 4 } { 3 5 7 9 } DOLSC 3 GET        | 2.
# El r de una recta perfecta da .999999999997, no 1: el programa calcula r por
# la misma secuencia que DCOVCORR (dividiendo por N-1 en cada paso) en vez del
# atajo SXY/RAIZ(SXX*SYY), a proposito, para que los dos den el MISMO numero.
# El precio es este arrastre de redondeo. Se prueba con tolerancia, que es lo
# que corresponde: lo que importa es la aritmetica, no el ultimo digito.
ols-perfecta-r  | { 1 2 3 4 } { 3 5 7 9 } DOLSC 4 GET 1 - ABS .000000001 <  | 1.
# Imperfecta, calculada aparte en la PC: r = .8528028654224, R2 = .7272727272727
ols-imperf-b0   | { 1 2 3 4 5 } { 2 4 5 4 6 } DOLSC 2 GET    | 1.8
ols-imperf-b1   | { 1 2 3 4 5 } { 2 4 5 4 6 } DOLSC 3 GET    | .8
ols-imperf-r    | { 1 2 3 4 5 } { 2 4 5 4 6 } DOLSC 4 GET .8528028654224 - ABS .000000001 < | 1.
ols-imperf-r2   | { 1 2 3 4 5 } { 2 4 5 4 6 } DOLSC 5 GET .7272727272727 - ABS .000000001 < | 1.

# ---------- DFRECTABLAC: Datos K H Inicio -> { K H START FREQ MAT } ----------
# 20 valores armados a mano para dar 2,2,6,6,4 con K=5, ancho 5, inicio 10:
# el mismo patron que el ejercicio de horas trabajadas de la catedra.
frec-frecuencias| { 11 12 16 17 21 21 22 22 23 23 26 26 27 27 28 28 31 32 33 34 } 5 5 10 DFRECTABLAC 4 GET | [ 2. 2. 6. 6. 4. ]
# La matriz tiene que salir de K filas por 6 columnas.
frec-tamano     | { 11 12 16 17 21 21 22 22 23 23 26 26 27 27 28 28 31 32 33 34 } 5 5 10 DFRECTABLAC 5 GET SIZE | { 5. 6. }
# Con K=0 usa Sturges: 1 + 3.322*log(20) = 5.32 -> 5.
frec-sturges    | { 11 12 16 17 21 21 22 22 23 23 26 26 27 27 28 28 31 32 33 34 } 0 5 10 DFRECTABLAC 1 GET | 5.

# ---------- Los chicos ----------
pib-pc          | 1000 2 10 DPIBPPAC                         | { 500. 50. }
# Duplicar en 2 anios: (2)^(1/2)-1 = 41.4213562373% anual. La maquina calcula
# la potencia por exp/ln y devuelve 41.421356237, con un digito menos de
# precision: se prueba con tolerancia.
cagr-duplica    | 100 0 2 DTCAMBIOC 1 GET 41.4213562373 - ABS .00000001 < | 1.
# |Z| = 3 NO es atipico (el criterio es > 3, estricto).
z-borde         | 10 4 2 DCOMPZC                             | { 3. 0 }
z-atipico       | 11 4 2 DCOMPZC                             | { 3.5 1 }

# ---------- DMKTC: a b c d T -> { PE QE } ----------
# Valores del apunte de catedra (AGENTS.md seccion 6).
mkt-sin-imp     | 100 2 10 3 0 DMKTC                         | { 18. 64. }
mkt-con-imp     | 100 2 10 3 5 DMKTC                         | { 21. 58. }

# ---------- DCOMPFIRMC: CF b c P -> { Q PROFIT CIERRA } ----------
# Los tres casos del apunte: beneficio positivo, negativo y cierre.
firm-gana       | 20 3 1 43 DCOMPFIRMC                       | { 20. 380. 0 }
firm-pierde     | 20 3 1 11 DCOMPFIRMC                       | { 4. -4. 0 }
# P < b: cierra, y la perdida es exactamente el costo fijo.
firm-cierra     | 20 3 1 1 DCOMPFIRMC                        | { 0. -20. 1 }
