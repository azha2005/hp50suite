%%HP: T(3)A(R)F(.);
\<<
  @ DCOMPFIRM - empresa competitiva, con grafico CMg/CVMe/P.
  @ Interfaz: el calculo esta en DCOMPFIRMC, que toma CF b c P de la
  @ pila y devuelve { Q PROFIT CIERRA }. Aca quedan el formulario,
  @ el dibujo y el veredicto.
  "Empresa Competitiva: CT=CF+b*q+c*q^2"
  {
    { "CF (Costo Fijo):" "0" 0 }
    { "b (Costo Marg. Base):" "0" 0 }
    { "c (Coef. Cuadratico):" "0" 0 }
    { "P (Precio de Mercado):" "0" 0 }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP \-> CF B C P
    \<<
      CF B C P DCOMPFIRMC OBJ\-> DROP \-> Q PROFIT CIERRA
      \<<
        IF Q 0 == THEN 10. ELSE Q 2. * END \->NUM \-> QMAX
        \<<
          B 2. C * QMAX * + \->NUM \-> CMGMAX
          \<<
            IF P CMGMAX > THEN P ELSE CMGMAX END 1.1 * \->NUM \-> PMAX
            \<<
              ERASE
              @ XRNG/YRNG con el rango real (0 a QMAX, 0 a PMAX),
              @ dibujando con los valores directos: asi el cursor del
              @ modo interactivo muestra Cantidad y Costo/Precio reales.
              0 QMAX XRNG 0 PMAX YRNG
              
              @ Ejes
              0 0 R\->C QMAX 0 R\->C LINE
              0 0 R\->C 0 PMAX R\->C LINE
              
              @ CMg: (Q=0,B) a (Q=QMAX,CMGMAX)
              0 B R\->C QMAX CMGMAX R\->C LINE
              
              @ CVMe: (Q=0,B) a (Q=QMAX,AVCMAX)
              B C QMAX * + \->NUM \-> AVCMAX
              \<<
                0 B R\->C QMAX AVCMAX R\->C LINE
                
                @ Linea horizontal de Precio
                0 P R\->C QMAX P R\->C LINE
                
                @ Punto de equilibrio (solo si no cerro), con offset
                @ proporcional a la escala de cada eje.
                IF Q 0 > THEN
                  QMAX 40. / \-> QOFFS
                  \<<
                    PMAX 40. / \-> POFFS
                    \<<
                      Q QOFFS - P R\->C Q QOFFS + P R\->C LINE
                      Q P POFFS - R\->C Q P POFFS + R\->C LINE
                    \>>
                  \>>
                END
                
                PICT PICTURE
                
                "Q*: " Q \->STR + 10 CHR +
                "Beneficio: " + PROFIT \->STR +
                MSGBOX
                IF CIERRA 1 == THEN
                  @ La semilla NO lleva + adelante: el MSGBOX de arriba ya
                  @ consumio el string anterior y la pila quedo vacia, asi
                  @ que el + no tenia con que concatenar (CLAUDE.md 2.4).
                  @ Estaba asi desde antes de partir el programa.
                  "La empresa CIERRA en el corto" 10 CHR +
                  "plazo (P menor al CMg minimo)." +
                  MSGBOX
                ELSE
                  "La empresa sigue produciendo" 10 CHR +
                  "en el corto plazo." +
                  MSGBOX
                END
              \>>
            \>>
          \>>
        \>>
      \>>
    \>>
  END
\>>
