%%HP: T(3)A(R)F(.);
DIR
  DCOMPFIRM
  \<<
    "Empresa Competitiva: CT=CF+b*q+c*q^2"
    {
      { "CF (Costo Fijo):" "0" 0 }
      { "b (Costo Marg. Base):" "0" 0 }
      { "c (Coef. Cuadratico):" "0" 0 }
      { "P (Precio de Mercado):" "0" 0 }
    }
    { } { } { }
    INFORM
    IF THEN
      OBJ\-> DROP \-> CF B C P
      \<<
        @ Con CT=CF+b*q+c*q^2, el costo marginal es CMg=b+2*c*q;
        @ el costo variable medio (CVMe=b+c*q) tiene su minimo
        @ tambien en q=0 con valor b -- por eso la condicion de
        @ cierre es simplemente P<b. Se calcula Q y Beneficio
        @ como expresiones IF/THEN/ELSE (no declaraciones dentro
        @ de ramas separadas), para que ambos queden disponibles
        @ despues y se pueda graficar en un solo lugar.
        IF P B < THEN 0 ELSE P B - 2. C * / END \->NUM \-> Q
        \<<
          IF P B < THEN
            0 CF -
          ELSE
            P Q * CF B Q * + C Q Q * * + -
          END \->NUM \-> PROFIT
          \<<
            IF Q 0 == THEN 10. ELSE Q 2. * END \->NUM \-> QMAX
            \<<
              B 2. C * QMAX * + \->NUM \-> CMGMAX
              \<<
                IF P CMGMAX > THEN P ELSE CMGMAX END 1.1 * \->NUM \-> PMAX
                \<<
                ERASE
                @ AGREGADO (version 2): XRNG/YRNG con el rango real
                @ (0 a QMAX, 0 a PMAX), dibujando con los valores
                @ directos -- el cursor del modo interactivo ahora
                @ muestra Cantidad y Costo/Precio reales.
                0 QMAX XRNG 0 PMAX YRNG
                
                @ Ejes
                0 0 R\->C QMAX 0 R\->C LINE
                0 0 R\->C 0 PMAX R\->C LINE
                
                @ CMg: (Q=0,B) a (Q=QMAX,CMGMAX)
                0 B R\->C QMAX CMGMAX R\->C LINE
                
                @ CVMe: (Q=0,B) a (Q=QMAX,AVCMAX)
                B C QMAX * + \->NUM \-> AVCMAX
                \<<
                  0 B R\->C QMAX AVCMAX R\->C LINE
                  
                  @ Linea horizontal de Precio
                  0 P R\->C QMAX P R\->C LINE
                  
                  @ Punto de equilibrio (solo si no cerro), con
                  @ offset proporcional a la escala de cada eje
                  IF Q 0 > THEN
                    QMAX 40. / \-> QOFFS
                    \<<
                      PMAX 40. / \-> POFFS
                      \<<
                        Q QOFFS - P R\->C Q QOFFS + P R\->C LINE
                        Q P POFFS - R\->C Q P POFFS + R\->C LINE
                      \>>
                    \>>
                  END
                  
                  PICT PICTURE
                  
                  "Q*: " Q \->STR + 10 CHR +
                  "Beneficio: " + PROFIT \->STR +
                  MSGBOX
                  IF P B < THEN
                    "La empresa CIERRA en el corto" + 10 CHR +
                    "plazo (P menor al CMg minimo)." +
                    MSGBOX
                  ELSE
                    "La empresa sigue produciendo" + 10 CHR +
                    "en el corto plazo." +
                    MSGBOX
                  END
                \>>
                \>>
              \>>
            \>>
          \>>
        \>>
      \>>
    END
  \>>
END
