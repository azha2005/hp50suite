%%HP: T(3)A(R)F(.);
\<<
  @ DPCTIL - percentil interpolado, formula de catedra Lp=(p/100)(n+1).
  @ La mediana es el percentil 50.
  @ Interfaz: el calculo esta en DPCTILC, que devuelve { N LP VAL }.
  "Percentil Interpolado (Catedra)"
  {
    { "Datos:" "Vector [x] o Lista {x}" }
    { "Percentil p (0-100):" "50" 0 }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP \-> DATA0 P
    \<<
      DATA0 P DPCTILC OBJ\-> DROP \-> N LP VAL
      \<<
        "P: " P \->STR + 10 CHR +
        "N: " + N \->STR +
        MSGBOX
        "Lp: " LP \->STR + 10 CHR +
        "Valor: " + VAL \->STR +
        MSGBOX
      \>>
    \>>
  END
\>>
