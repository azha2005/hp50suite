%%HP: T(3)A(R)F(.);
\<<
  @ DCONT - continuidad de una funcion a trozos en el punto de corte.
  @ Cubre entero el punto 2 de la guia (los cuatro incisos).
  @
  @ Este objeto es SOLO la interfaz: formulario, cajas y veredicto.
  @ La matematica esta en DCONTC, que toma 'FI' 'FD' A de la pila y
  @ devuelve { LI LD IG ES }. Estan separados para que el harness
  @ pueda probar el calculo sin poder tipear en un INFORM.
  @
  @ Los dos limites quedan tambien en la pila, por si el texto no entra
  @ en la caja.
  "Continuidad en el corte"
  {
    { "Rama izquierda, ej 'X^2':" "'X'" }
    { "Rama derecha, ej '2*X':" "'X'" }
    { "Punto de corte a:" "0" 0 }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP DCONTC OBJ\-> DROP \-> LI LD IG ES
    \<<
      "Lim izq: " LI \->STR + 10 CHR +
      "Lim der: " + LD \->STR +
      MSGBOX
      IF ES 0 >
      THEN
        "ESENCIAL (2a esp.)" 10 CHR +
        "Asint. vertical" +
      ELSE
        IF IG
        THEN
          "LIMITES IGUALES" 10 CHR +
          "Ver f(a): evitable?" +
        ELSE
          "SALTO FINITO" 10 CHR +
          "(1a especie)" +
        END
      END
      MSGBOX
      LI LD
    \>>
  END
\>>
