\<<
  \-> F0 A
  \<<
    RCLF \-> FL
    \<<
      -105 CF -103 CF -3 CF -120 SF -123 CF -119 CF -100 CF -99 CF RAD
      DEPTH \-> D0
      \<<
        "" { } "ERROR" \-> VAL TR AV
        \<<
          IFERR
IF { X Y } DFREEC THEN "X/Y ASIGNADA" DOERR END
F0 DREADC \-> F
\<<
 F 'X' A DVALC \-> FA
 \<<
  IF FA TYPE 2 == THEN "PUNTO FUERA DE DOMINIO" DOERR END
  TR "Punto de tangencia: (a,f(a))" A FA 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO
  F 'X' DDERC \-> DR
  \<<
   TR DR 2 GET + 'TR' STO
   IF DR 3 GET "OK" == NOT THEN DR 3 GET DOERR END
   F F A FA DDERLATC \-> LR
   \<<
    TR LR 2 GET + 'TR' STO
    IF LR 3 GET "OK" == NOT THEN LR 3 GET DOERR END
    LR 1 GET 1 GET LR 1 GET 2 GET \-> ML MR
    \<<
     "" "" "" \-> M TG NR
     \<<
      IF LR 1 GET 3 GET "DERIVABLE" == THEN
       ML \->NUM 'M' STO
       TR "Pendiente: m=f'(a)" M 2 \->LIST 1 \->LIST + 'TR' STO
       TR "Tangente: Y-f(a)=m*(X-a)" 'Y' FA - M 'X' A - * = 2 \->LIST 1 \->LIST + 'TR' STO
       'Y' FA \->NUM M 'X' A - * + EXPAND = 'TG' STO
       IF M 0 == THEN
        'X' A = 'NR' STO
        TR "m=0: normal vertical X=a" NR 2 \->LIST 1 \->LIST + 'TR' STO
       ELSE
        TR "Normal perpendicular: mn=-1/m" -1 M / 2 \->LIST 1 \->LIST + 'TR' STO
        'Y' FA \->NUM -1 M / 'X' A - * + EXPAND = 'NR' STO
       END
      ELSE
       IF ML DLKINDC 2 == MR DLKINDC 2 == AND THEN
        "PENDIENTE INFINITA" 'M' STO 'X' A = 'TG' STO 'Y' FA = 'NR' STO
        TR "Ambos cocientes infinitos: tangente vertical" TG 2 \->LIST 1 \->LIST + 'TR' STO
       ELSE "NO HAY TANGENTE UNICA" DOERR END
      END
      TR "Tangente final" TG 2 \->LIST 1 \->LIST + 'TR' STO
      TR "Normal final" NR 2 \->LIST 1 \->LIST + 'TR' STO
      M FA \->NUM TG NR 4 \->LIST 'VAL' STO "OK" 'AV' STO
     \>>
    \>>
   \>>
  \>>
