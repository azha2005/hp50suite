%%HP: T(3)A(R)F(.);
\<<
  \-> E0 V
  \<<
    RCLF \-> FL
    \<<
      -105 CF -103 CF -3 CF -120 SF -123 CF -119 CF -100 CF -99 CF RAD
      DEPTH \-> D0
      \<<
        "" { } "ERROR" \-> VAL TR AV
        \<<
          IFERR
E0 DREADC \-> E
\<<
  TR "Derivar respecto de" V 2 \->LIST 1 \->LIST + 'TR' STO
  TR "Expresion a derivar" E 2 \->LIST 1 \->LIST + 'TR' STO
  IF E TYPE 9 == THEN
    E OBJ\-> \->STR \-> N OP
    \<<
      N \->LIST \-> AA
      \<<
        { } \-> DD
        \<<
          1 AA SIZE FOR I
            AA I GET V DDERC \-> RR
            \<<
              TR RR 2 GET + 'TR' STO
              IF RR 3 GET "OK" == NOT THEN RR 3 GET DOERR END
              DD RR 1 GET 1 \->LIST + 'DD' STO
            \>>
          NEXT
          AA 1 GET DD 1 GET 0 0 \-> A DA B DB
          \<<
            IF N 2 == THEN AA 2 GET 'B' STO DD 2 GET 'DB' STO END
            "" "" \-> RULE FORM
            \<<
              CASE
                OP "+" == THEN "Suma: (a+b)'=a'+b'" 'RULE' STO DA DB + 'FORM' STO END
                OP "-" == THEN "Resta: (a-b)'=a'-b'" 'RULE' STO DA DB - 'FORM' STO END
                OP "*" == THEN "Producto: a'b+ab'" 'RULE' STO DA B * A DB * + 'FORM' STO END
                OP "/" == THEN "Cociente: (a'b-ab')/b^2; b<>0" 'RULE' STO DA B * A DB * - B SQ / 'FORM' STO END
                OP "^" == THEN
                  IF DB 0 SAME THEN
                    "Potencia y cadena: b*a^(b-1)*a'" 'RULE' STO
                    IF B 0 SAME THEN 0 ELSE B A B 1 - ^ * DA * END 'FORM' STO
                  ELSE
                    "a^b: a>0; a^b*(b'*LN(a)+b*a'/a)" 'RULE' STO
                    A B ^ DB A LN * B DA * A / + * 'FORM' STO
                  END
                END
                OP "NEG" == THEN "(-a)'=-a'" 'RULE' STO DA NEG 'FORM' STO END
                OP "INV" == THEN "(1/a)'=-a'/a^2; a<>0" 'RULE' STO DA NEG A SQ / 'FORM' STO END
                OP "SQ" == THEN "(a^2)'=2*a*a'" 'RULE' STO 2 A * DA * 'FORM' STO END
                OP "EXP" == THEN "EXP(a)'=EXP(a)*a'" 'RULE' STO A EXP DA * 'FORM' STO END
                OP "LN" == THEN "LN(a)'=a'/a; a>0" 'RULE' STO DA A / 'FORM' STO END
                OP "LOG" == THEN "LOG(a)'=a'/(a*LN(10)); a>0" 'RULE' STO DA A 10 LN * / 'FORM' STO END
                OP "SIN" == THEN "SIN(a)'=COS(a)*a'" 'RULE' STO A COS DA * 'FORM' STO END
                OP "COS" == THEN "COS(a)'=-SIN(a)*a'" 'RULE' STO A SIN NEG DA * 'FORM' STO END
                OP "TAN" == THEN "TAN(a)'=a'/COS(a)^2; COS(a)<>0" 'RULE' STO DA A COS SQ / 'FORM' STO END
                OP "ASIN" == THEN "ASIN(a)'=a'/sqrt(1-a^2); |a|<1" 'RULE' STO DA 1 A SQ - \v/ / 'FORM' STO END
                OP "ACOS" == THEN "ACOS(a)'=-a'/sqrt(1-a^2); |a|<1" 'RULE' STO DA NEG 1 A SQ - \v/ / 'FORM' STO END
                OP "ATAN" == THEN "ATAN(a)'=a'/(1+a^2)" 'RULE' STO DA 1 A SQ + / 'FORM' STO END
                OP "ABS" == THEN "ABS(a)'=a*a'/ABS(a); a<>0; estudiar ceros aparte" 'RULE' STO A DA * A ABS / 'FORM' STO END
                OP 131 CHR == THEN "sqrt(a)'=a'/(2*sqrt(a)); a>0" 'RULE' STO DA 2 A \v/ * / 'FORM' STO END
                OP "XROOT" == THEN
                  "Raiz: a es indice constante; (b^(1/a))'" 'RULE' STO
                  IF DB 0 SAME NOT THEN "INDICE VARIABLE: USAR POTENCIA" DOERR END
                  B INV A B INV 1 - ^ * DA * 'FORM' STO
                END
                "REGLA NO SOPORTADA" DOERR
              END
              TR "Regla aplicada" RULE 2 \->LIST 1 \->LIST + 'TR' STO
              TR "Interior a, derivada a'" A DA 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO
              IF N 2 == THEN TR "b, derivada b'" B DB 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO END
              TR "Sustituir en la regla" FORM 2 \->LIST 1 \->LIST + 'TR' STO
              FORM EXPAND SIMPLIFY 'VAL' STO
            \>>
          \>>
        \>>
      \>>
    \>>
  ELSE
    IF E V SAME THEN 1 "Variable: dV/dV=1" ELSE 0 "Constante respecto de V: derivada 0" END
    \-> D RULE
    \<< D 'VAL' STO TR "Regla" RULE 2 \->LIST 1 \->LIST + 'TR' STO \>>
  END
  TR "Simplificar derivada" VAL 2 \->LIST 1 \->LIST + 'TR' STO
  "OK" 'AV' STO
\>>
          THEN "" 'VAL' STO ERRM 'AV' STO END
          TR "Estado" AV 2 \->LIST 1 \->LIST + 'TR' STO
          VAL TR AV 3 \->LIST
          \-> RES \<< DEPTH D0 - DROPN FL STOF RES \>>
        \>>
      \>>
    \>>
  \>>
\>>
