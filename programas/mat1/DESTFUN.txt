\<<
  @ DESTFUN - estudio de funcion, hasta donde llegue el CAS.
  @ Este objeto es SOLO la interfaz: el calculo esta en DESTFUNC, que
  @ toma 'f(X)' AV PV de la pila y devuelve { D1 Z1 D2 Z2 TP TN TV }.
  @ Las asintotas las calcula DASHOC, no una copia: antes esta logica
  @ estaba escrita dos veces y el bug del infinito hubo que arreglarlo
  @ dos veces (la segunda copia se habia quedado sin arreglar).
  @ NO calcula dominio ni imagen: decision tomada, ver SUITE-MAT1 4.
  @
  @ REQUIERE el global INF (correr DINF una vez, ver SUITE-MAT1 2).
  @
  @ Deja en la pila n4 = f', n3 = ceros de f', n2 = f'', n1 = ceros de f''.
  IF INF TYPE 9 == NOT
  THEN
    "Falta el global INF" 10 CHR +
    "Correr DINF primero" +
    MSGBOX
  ELSE
    "Estudio de funcion"
    {
      { "f(X):" "'X^3-3*X'" }
      { "Candidato asint. vertical:" "0" 0 }
      { "Probar vertical? 1=si 0=no:" "0" 0 }
    }
    { } { } { }
    INFORM
    IF THEN
      OBJ\-> DROP \-> F0 AV PV
      \<<
        F0 AV PV DESTFUNC OBJ\-> DROP \-> D1 Z1 D2 Z2 TP TN TV
        \<<
          "f'(X) =" 10 CHR + D1 \->STR + MSGBOX
          "Criticos (f'=0):" 10 CHR + Z1 \->STR + MSGBOX
          "f''(X) =" 10 CHR + D2 \->STR + MSGBOX
          "Inflexion (f''=0):" 10 CHR + Z2 \->STR + MSGBOX
          "En +oo:" 10 CHR + TP + MSGBOX
          "En -oo:" 10 CHR + TN + MSGBOX
          IF PV 1 ==
          THEN
            "En X=" AV \->STR + ":" + 10 CHR + TV + MSGBOX
          END
          "Detalle completo" 10 CHR +
          "en la pila (4 niveles)" +
          MSGBOX
          D1 Z1 D2 Z2
        \>>
      \>>
    END
  END
\>>
