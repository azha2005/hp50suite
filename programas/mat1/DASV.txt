%%HP: T(3)A(R)F(.);
\<<
  @ DASV - asintota vertical en un punto candidato.
  @ Este objeto es SOLO la interfaz: el calculo esta en DASVC, que toma
  @ 'f(X)' A de la pila y devuelve { VI VD HAY }.
  @ No busca los candidatos solo: salen de anular denominadores o de los
  @ bordes del dominio, y eso lo pone el usuario.
  @ Los dos valores sondeados quedan tambien en la pila.
  "Asintota vertical"
  {
    { "f(X), ej '1/X':" "'1/X'" }
    { "Punto candidato a:" "0" 0 }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP \-> F0 A
    \<<
      F0 A DASVC OBJ\-> DROP \-> VI VD HAY
      \<<
        "f(a-) = " VI \->STR + 10 CHR +
        "f(a+) = " + VD \->STR +
        MSGBOX
        IF HAY 1 ==
        THEN
          "HAY ASINTOTA" 10 CHR +
          "vertical en X=" + A \->STR +
        ELSE
          IF HAY 0 ==
          THEN
            "SIN ASINTOTA" 10 CHR +
            "Queda acotada cerca" +
          ELSE
            "NO SE PUDO EVALUAR" 10 CHR +
            "a los dos lados" +
          END
        END
        MSGBOX
        VI VD
      \>>
    \>>
  END
\>>
