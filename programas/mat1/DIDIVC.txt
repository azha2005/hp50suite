%%HP: T(3)A(R)F(.);
\<<
  \-> N D V
  \<<
    RCLF \-> FL
    \<<
      -105 CF -103 CF -3 CF -120 SF -123 CF -119 CF -100 CF -99 CF RAD
      DEPTH \-> D0
      \<<
        "" { } "ERROR" \-> VAL TR AV
        \<<
          IFERR
D V DERIV SIMPLIFY \-> DD
\<<
 IF DD 0 SAME THEN
  N V DIPASC \-> RR
  \<< IF RR 3 GET "OK" == NOT THEN RR 3 GET DOERR END
   TR RR 2 GET + 'TR' STO RR 1 GET D / SIMPLIFY 'VAL' STO
   TR "Dividir por constante" D 2 \->LIST 1 \->LIST + 'TR' STO
  \>>
 ELSE
  N DD / SIMPLIFY \-> Q
  \<<
   IF Q V DERIV SIMPLIFY 0 SAME THEN
    TR "u=denominador" D 2 \->LIST 1 \->LIST + 'TR' STO
    TR "du/dV" DD 2 \->LIST 1 \->LIST + 'TR' STO
    TR "Ajuste N=q*du/dV; integral q du/u" Q 2 \->LIST 1 \->LIST + 'TR' STO
    Q D ABS LN * SIMPLIFY 'VAL' STO
   ELSE
    D 1 0 \-> BASE POW IE
    \<<
     IF D TYPE 9 == THEN
      D OBJ\-> \->STR \-> NA OP
      \<< NA \->LIST \-> AR \<<
       IF OP "^" == THEN AR 1 GET 'BASE' STO AR 2 GET 'POW' STO END
       IF OP 131 CHR == THEN AR 1 GET 'BASE' STO 1 2 / 'POW' STO END
       IF OP "XROOT" == THEN AR 1 GET 'BASE' STO AR 2 GET INV 'POW' STO END
       IF OP "EXP" == THEN AR 1 GET NEG EXP 'BASE' STO 1 'IE' STO END
      \>> \>>
     END
     IF IE THEN
      N BASE V DSUB1C \-> ER
      \<<
       TR "Reciproco exponencial: 1/EXP(h)=EXP(-h)" BASE 2 \->LIST 1 \->LIST + ER 2 GET + 'TR' STO
       IF ER 3 GET "OK" == NOT THEN ER 3 GET DOERR END
       ER 1 GET 'VAL' STO
      \>>
     ELSE
     BASE V DERIV SIMPLIFY \-> DB
     \<<
      IF DB V DERIV SIMPLIFY 0 SAME DB 0 SAME NOT AND N V DERIV SIMPLIFY 0 SAME AND THEN
       TR "Potencia del denominador lineal: u=base" BASE 2 \->LIST 1 \->LIST + 'TR' STO
       TR "du/dV y potencia p" DB POW 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO
       IF POW 1 SAME THEN N DB / BASE ABS LN * ELSE N BASE 1 POW - ^ * DB 1 POW - * / END SIMPLIFY 'VAL' STO
       TR "Integral u^(-p); dividir por pendiente" VAL 2 \->LIST 1 \->LIST + 'TR' STO
      ELSE
       N BASE POW V DIQUAC \-> RR
       \<<
        TR RR 2 GET + 'TR' STO
        IF RR 3 GET "OK" == NOT THEN RR 3 GET DOERR END
        RR 1 GET 'VAL' STO
       \>>
      END
     \>>
     END
    \>>
   END
  \>>
 END
 "OK" 'AV' STO
\>>
          THEN "" 'VAL' STO ERRM 'AV' STO END
          TR "Estado" AV 2 \->LIST 1 \->LIST + 'TR' STO
          VAL TR AV 3 \->LIST
          \-> RES \<< DEPTH D0 - DROPN FL STOF RES \>>
        \>>
      \>>
    \>>
  \>>
\>>
