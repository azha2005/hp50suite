%%HP: T(3)A(R)F(.);
\<<
  @ F a b -> { candidates values xmax fmax xmin fmin status
  @            ALLmax ALLmin trace }. New implementation behind DWEIERC.
  @ Invalid hypotheses -> NO extrema claimed. Constant -> all points of [a,b].
  \-> F0 A0 B0
  \<<
    RCLF \-> FL
    \<<
      -105 CF -103 CF -3 CF -120 SF -123 CF -119 CF -100 CF -99 CF RAD
      DEPTH \-> D0
      \<<
        { } { } "" "" "" "" "ERROR" { } { } { } \-> CN VL XMX FMX XMN FMN AV AM AX TR
        \<<
          IFERR
            IF { X Y U } DFREEC THEN "X/Y/U ASIGNADA" DOERR END
            F0 DREADC A0 'X' 0 DVALC B0 'X' 0 DVALC \-> F AA BB
            \<<
              IF AA TYPE 2 == BB TYPE 2 == OR THEN "EXTREMOS FINITOS REALES" DOERR END
              IF AA \->NUM BB \->NUM > THEN BB AA ELSE AA BB END \-> A B
              \<<
                TR "Cerrado y acotado [a,b]" A B 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO
                F A B DCLOSEC \-> HC
                \<<
                  TR "Condiciones de dominio original" HC 3 GET 2 \->LIST 1 \->LIST + 'TR' STO
                  TR "Continuidad por composicion y condiciones" HC 1 GET 2 \->LIST 1 \->LIST + 'TR' STO
                  IF HC 1 GET "OK" == NOT THEN HC 1 GET DOERR END
                  F 1 DBRKC \-> ST
                  \<<
                    IF ST 5 GET SIZE THEN ST 5 GET DOERR END
                    TR "f'(X)" F 'X' DERIV SIMPLIFY 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Ceros de f'" ST 1 GET 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "No derivables: incluir si f existe" ST 4 GET 2 \->LIST 1 \->LIST + 'TR' STO
                    A B 2 \->LIST 'CN' STO
                    ST 1 GET ST 4 GET + \-> ROWS
                    \<<
                      IF ROWS SIZE 0 > THEN
                        1 ROWS SIZE FOR I
                          ROWS I GET \-> P
                          \<<
                            IF P \->NUM A \->NUM > P \->NUM B \->NUM < AND THEN CN P 1 \->LIST + 'CN' STO END
                          \>>
                        NEXT
                      END
                    \>>
                    CN DORDC 'CN' STO
                    1 CN SIZE FOR I
                      CN I GET \-> P
                      \<<
                        F 'X' P DVALC \-> FV
                        \<<
                          IF FV TYPE 2 == THEN "CANDIDATO FUERA DE DOMINIO" DOERR END
                          VL FV 1 \->LIST + 'VL' STO
                          TR "Evaluar candidato: x,f(x)" P FV 2 \->LIST 2 \->LIST 1 \->LIST + 'TR' STO
                          IF I 1 == THEN P 'XMX' STO FV 'FMX' STO P 'XMN' STO FV 'FMN' STO END
                          IF FV \->NUM FMX \->NUM > THEN P 'XMX' STO FV 'FMX' STO END
                          IF FV \->NUM FMN \->NUM < THEN P 'XMN' STO FV 'FMN' STO END
                        \>>
                      \>>
                    NEXT
                    1 CN SIZE FOR I
                      IF VL I GET FMX - SIMPLIFY 0 == THEN AX CN I GET 1 \->LIST + 'AX' STO END
                      IF VL I GET FMN - SIMPLIFY 0 == THEN AM CN I GET 1 \->LIST + 'AM' STO END
                    NEXT
                    IF F 'X' DERIV SIMPLIFY 0 SAME A B == NOT AND THEN
                      "TODO [a,b]" 'AX' STO "TODO [a,b]" 'AM' STO
                    END
                    TR "Comparar valores: maximo" FMX 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Todos los puntos de maximo" AX 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Comparar valores: minimo" FMN 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Todos los puntos de minimo" AM 2 \->LIST 1 \->LIST + 'TR' STO
                    "OK" 'AV' STO
                  \>>
                \>>
              \>>
            \>>
          THEN
            ERRM 'AV' STO "" 'XMX' STO "" 'FMX' STO "" 'XMN' STO "" 'FMN' STO { } 'AX' STO { } 'AM' STO
          END
          TR "Estado" AV 2 \->LIST 1 \->LIST + 'TR' STO
          CN VL XMX FMX XMN FMN AV AX AM TR 10 \->LIST
          \-> RES \<< DEPTH D0 - DROPN RES \>>
        \>>
      \>>
      FL STOF
    \>>
  \>>
\>>
