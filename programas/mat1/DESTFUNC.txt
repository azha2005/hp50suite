%%HP: T(3)A(R)F(.);
\<<
  @ F AV PV -> seven legacy fields plus the full DANAC result in field 8.
  \-> F0 AV PV
  \<<
    RCLF \-> FL
    \<<
    -105 CF -103 CF -3 CF -119 CF -120 SF -123 CF -100 CF RAD
    DEPTH \-> D0
    \<<
      F0 DANAC \-> ST
      \<<
        "INF requerido para asintotas" "INF requerido para asintotas" "no probada" ST 11 GET \-> TP TN TV TR
        \<<
          IF ST 12 GET "OK" == THEN
          IF { INF } DFREEC THEN
            IFERR F0 DREADC DASHOC THEN { 0 0 0 0 "no se pudo" "no se pudo" } END
            \-> ASY
            \<<
              ASY 5 GET 'TP' STO ASY 6 GET 'TN' STO
              TR "m(+oo)=lim f(X)/X" ASY 1 GET 2 \->LIST 1 \->LIST + 'TR' STO
              TR "b(+oo)=lim (f(X)-m*X)" ASY 2 GET 2 \->LIST 1 \->LIST + 'TR' STO
              TR "En +oo" TP 2 \->LIST 1 \->LIST + 'TR' STO
              TR "m(-oo)=lim f(X)/X" ASY 3 GET 2 \->LIST 1 \->LIST + 'TR' STO
              TR "b(-oo)=lim (f(X)-m*X)" ASY 4 GET 2 \->LIST 1 \->LIST + 'TR' STO
              TR "En -oo" TN 2 \->LIST 1 \->LIST + 'TR' STO
            \>>
          ELSE
            TR "Asintotas horizontales/oblicuas" TP 2 \->LIST 1 \->LIST + 'TR' STO
          END
          ST 5 GET 3 GET
          IF PV 1 == THEN AV 1 \->LIST + END DORDC \-> CP
          \<<
            IF CP SIZE 0 > THEN
              1 CP SIZE FOR I
                CP I GET \-> P
                \<<
                  DEPTH \-> D1
                  \<<
                    IFERR F0 DREADC P DCONTFC THEN { "?" "?" "?" -1 } END
                    \-> CR \<< DEPTH D1 - DROPN CR \>>
                  \>>
                  \-> CR
                  \<<
                    TR "Candidato vertical x=" P 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Limite izquierdo (real)" CR 2 GET 2 \->LIST 1 \->LIST + 'TR' STO
                    TR "Limite derecho (real)" CR 3 GET 2 \->LIST 1 \->LIST + 'TR' STO
                    IF CR 4 GET 4 == THEN "HAY ASINT. VERTICAL"
                    ELSE IF CR 4 GET -1 == THEN "no se pudo decidir" ELSE "sin lateral infinito" END END
                    \-> VER
                    \<<
                      TR "Conclusion vertical" VER 2 \->LIST 1 \->LIST + 'TR' STO
                      IF PV 1 == THEN IF P \->NUM AV \->NUM == THEN VER 'TV' STO END END
                    \>>
                  \>>
                \>>
              NEXT
            END
          \>>
          ELSE
            "Estudio no completado: revisar estado" 'TP' STO TP 'TN' STO
          END
          ST 11 TR PUT 'ST' STO
          ST 1 GET ST 2 GET ST 3 GET ST 4 GET TP TN TV ST 8 \->LIST
          \-> RES \<< DEPTH D0 - DROPN RES \>>
        \>>
      \>>
    \>>
    FL STOF
    \>>
  \>>
\>>
