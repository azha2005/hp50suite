%%HP: T(3)A(R)F(.);
\<<
  @ DDOMP - sondea el dominio de F(X) en una lista de puntos.
  @ Confirma que puntos pertenecen; NO deduce el dominio.
  @ Requiere modo Complex APAGADO: si no, la raiz de un negativo
  @ devuelve un complejo en vez de error y el punto pasa por bueno.
  @ La lista de resultados (valor o "FUERA") queda en la pila.
  "Sondear dominio"
  {
    { "F(X), ej '\v/X':" "'\v/X'" }
    { "Puntos a probar:" "{ -2 -1 0 1 2 4 }" }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP \-> F0 P0
    \<<
      F0 IF DUP TYPE 2 == THEN STR\-> END \-> F
      \<<
        P0 IF DUP TYPE 2 == THEN STR\-> END \-> PTS
        \<<
          { } 0 \-> RES DENTRO
          \<<
            1 PTS SIZE FOR I
              PTS I GET 'X' STO
              IFERR
                F \->NUM DENTRO 1 + 'DENTRO' STO
              THEN
                "FUERA"
              END
              1 \->LIST RES SWAP + 'RES' STO
            NEXT
            IFERR { X } PURGE THEN 0 DROP END
            "Puntos probados: " PTS SIZE \->STR + 10 CHR +
            "En el dominio: " + DENTRO \->STR +
            MSGBOX
            "Detalle en la pila" 10 CHR +
            "(valor o FUERA)" +
            MSGBOX
            RES
          \>>
        \>>
      \>>
    \>>
  END
\>>
