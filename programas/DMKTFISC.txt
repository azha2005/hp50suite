%%HP: T(3)A(R)F(.);
\<<
  @ DMKTFISC - mercado con impuesto/subsidio, controles de precio y grafico.
  @ Tres capas: DMKTFISCC calcula los trece numeros, DMKTFISCG dibuja, y
  @ este objeto pone los dos formularios, el PICTURE y las seis cajas.
  @ PICTURE es interactivo, asi que queda aca: de otro modo el harness no
  @ puede probar el grafico.
  @ Van dos INFORM porque el limite practico es ~4 campos por pantalla.
  "Mercado + Impuestos: Qd=a-b*P, Qs=c+d*P"
  {
    { "a (Demanda Auto):" "0" 0 }
    { "b (Sensib. Dem):" "0" 0 }
    { "c (Oferta Auto):" "0" 0 }
    { "d (Sensib. Of):" "0" 0 }
  }
  { } { } { }
  INFORM
  IF THEN
    OBJ\-> DROP \-> A B C D
    \<<
      "Impuesto y Controles de Precio"
      {
        { "Impuesto T (0=sin imp):" "0" 0 }
        { "Precio Maximo (0=sin techo):" "0" 0 }
        { "Precio Minimo (0=sin piso):" "0" 0 }
      }
      { } { } { }
      INFORM
      IF THEN
        OBJ\-> DROP \-> T PMAXW PMINW
        \<<
          A B C D T PMAXW PMINW DMKTFISCC \-> RES
          \<<
            A B C D T PMAXW PMINW RES DMKTFISCG
            PICT PICTURE
            "Pb (comprador): " RES 1 GET \->STR + 10 CHR +
            "Ps (vendedor): " + RES 3 GET \->STR +
            MSGBOX
            "Q*: " RES 2 GET \->STR + 10 CHR +
            "Recaudacion: " + RES 8 GET \->STR +
            MSGBOX
            "PIE (DWL): " RES 9 GET \->STR + 10 CHR +
            "Q sin imp: " + RES 5 GET \->STR +
            MSGBOX
            "Inc.Comprador: " RES 10 GET \->STR + "%" + 10 CHR +
            "Inc.Vendedor: " + RES 11 GET \->STR + "%" +
            MSGBOX
            "Escasez (techo): " RES 6 GET \->STR + 10 CHR +
            "Excedente (piso): " + RES 7 GET \->STR +
            MSGBOX
            "Ed (demanda): " RES 12 GET \->STR + 10 CHR +
            "Es (oferta): " + RES 13 GET \->STR +
            MSGBOX
          \>>
        \>>
      END
    \>>
  END
\>>
